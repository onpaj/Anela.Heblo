```ts
export function filterPositions(
  positions: Position[],
  departmentFilter: string,
  levelFilter: string,
): Position[] {
  let result = positions;

  if (departmentFilter !== 'all') {
    const deptSet = new Set(
      positions
        .filter(p => p.department === departmentFilter)
        .flatMap(p => [p.id!, ...getAllParentPositionIds(p.id!, positions)])
    );
    result = positions.filter(p => deptSet.has(p.id!));
  }

  if (levelFilter !== 'all') {
    const maxLevel = parseInt(levelFilter, 10);
    result = result.filter(p => !p.level || p.level <= maxLevel);
  }

  return result;
}
```

Then in `OrgChartPage.tsx`:
```ts
const filteredPositions = useMemo(
  () => filterPositions(orgData.organization.positions, filters.department, filters.level),
  [orgData, filters],
);
```

This also converts the IIFE to a `useMemo` (avoiding re-computation on zoom/positionRect changes), and makes the logic unit-testable without rendering the component.
